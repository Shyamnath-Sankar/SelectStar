    29→/**
    30→ * ConnectionScreen — redesigned for a cleaner first-run experience.
    31→ *
    32→ * Single-column hero on top, prominent two-card mode picker below, then the
    33→ * mode-specific form (DB connection string OR file upload). Recent sessions
    34→ * collapse to a compact strip at the bottom. The goal: a user lands on the
    35→ * page, picks a mode in one click, fills one field, hits Connect.
    36→ */
    37→export function ConnectionScreen() {
    38→  const { connecting, connectError, connect, setConnecting, setConnectError, loadSession } = useSession();
    39→  const [mode, setMode] = useState<AppMode>("sql");
    40→  const [connStr, setConnStr] = useState("demo");
    41→  const [recent, setRecent] = useState<RecentSession[]>([]);
    42→  const [loadingSession, setLoadingSession] = useState<string | null>(null);
    43→  const [showRecent, setShowRecent] = useState(false);
    44→  const [uploadingFile, setUploadingFile] = useState<File | null>(null);
    45→  const [uploading, setUploading] = useState(false);
    46→  const [dragActive, setDragActive] = useState(false);
    47→
    48→  useEffect(() => {
    49→    void (async () => {
    50→      try {
    51→        const res = await fetch("/api/sessions");
    52→        const data = await res.json();
    53→        setRecent((data.sessions || []).filter((s: RecentSession) => s.status === "connected"));
    54→      } catch { /* ignore */ }
    55→    })();
    56→  }, []);
    57→
    58→  // ---- SQL mode: connect to a database via connection string -----------
    59→  async function handleConnect(e?: React.FormEvent) {
    60→    e?.preventDefault();
    61→    if (!connStr.trim()) {
    62→      toast.error("Please enter a connection string.");
    63→      return;
    64→    }
    65→    setConnecting(true);
    66→    setConnectError(null);
    67→    try {
    68→      const res = await fetch("/api/connect", {
    69→        method: "POST",
    70→        headers: { "Content-Type": "application/json" },
    71→        body: JSON.stringify({ connectionString: connStr.trim() }),
    72→      });
    73→      const data = await res.json();
    74→      if (!res.ok) {
    75→        setConnectError(data.error || "Connection failed");
    76→        toast.error(data.error || "Connection failed");
    77→        return;
    78→      }
    79→      connect({
    80→        sessionId: data.sessionId,
    81→        mode: "sql",
    82→        label: data.dialect === "sqlite" && /demo/i.test(connStr) ? "Demo E-commerce DB" : `${data.dialect} session`,
    83→        dialect: data.dialect,
    84→        schema: data.schema,
    85→        canWrite: data.canWrite,
    86→        suggestedQuestions: data.suggestedQuestions,
    87→      });
    88→      if (data.domain) useSession.getState().setDomain(data.domain);
    89→      toast.success(`Connected — ${data.schema.tables.length} tables found.`);
    90→    } catch (e) {
    91→      setConnectError((e as Error).message);
    92→      toast.error((e as Error).message);
    93→    } finally {
    94→      setConnecting(false);
    95→    }
    96→  }
    97→
    98→  // ---- Classic mode: upload CSV/XLSX file(s) ---------------------------
    99→  async function handleUpload(file: File) {
   100→    setUploadingFile(file);
   101→    setUploading(true);
   102→    setConnectError(null);
   103→    try {
   104→      const fd = new FormData();
   105→      fd.append("file", file);
   106→      const res = await fetch("/api/classic/upload", { method: "POST", body: fd });
   107→      const data = await res.json();
   108→      if (!res.ok) {
   109→        setConnectError(data.error || "Upload failed");
   110→        toast.error(data.error || "Upload failed");
   111→        return;
   112→      }
   113→      connect({
   114→        sessionId: data.sessionId,
   115→        mode: "classic",
   116→        label: data.label,
   117→        dialect: data.dialect,
   118→        schema: data.schema,
   119→        canWrite: true,
   120→        suggestedQuestions: data.suggestedQuestions,
   121→      });
   122→      if (data.domain) useSession.getState().setDomain(data.domain);
   123→      const tableCount = data.schema.tables.length;
   124→      const totalRows = data.schema.tables.reduce((s: number, t: { rowCount: number }) => s + t.rowCount, 0);
   125→      toast.success(`Loaded ${tableCount} ${tableCount === 1 ? "table" : "tables"} · ${totalRows.toLocaleString()} rows from ${file.name}.`);
   126→    } catch (e) {
   127→      setConnectError((e as Error).message);
   128→      toast.error((e as Error).message);
   129→    } finally {
   130→      setUploading(false);
   131→    }
   132→  }
   133→
   134→  // Multi-file upload — accept an array of files.
   135→  async function handleUploadMultiple(files: FileList | File[]) {
   136→    const arr = Array.from(files);
   137→    if (!arr.length) return;
   138→    setUploadingFile(arr[0]);
   139→    setUploading(true);
   140→    setConnectError(null);
   141→    try {
   142→      const fd = new FormData();
   143→      for (const f of arr) fd.append("files", f);
   144→      const res = await fetch("/api/classic/upload", { method: "POST", body: fd });
   145→      const data = await res.json();
   146→      if (!res.ok) {
   147→        setConnectError(data.error || "Upload failed");
   148→        toast.error(data.error || "Upload failed");
   149→        return;
   150→      }
   151→      connect({
   152→        sessionId: data.sessionId,
   153→        mode: "classic",
   154→        label: data.label,
   155→        dialect: data.dialect,
   156→        schema: data.schema,
   157→        canWrite: true,
   158→        suggestedQuestions: data.suggestedQuestions,
   159→      });
   160→      if (data.domain) useSession.getState().setDomain(data.domain);
   161→      const tableCount = data.schema.tables.length;
   162→      const totalRows = data.schema.tables.reduce((s: number, t: { rowCount: number }) => s + t.rowCount, 0);
   163→      toast.success(`Loaded ${tableCount} ${tableCount === 1 ? "table" : "tables"} · ${totalRows.toLocaleString()} rows from ${arr.length} ${arr.length === 1 ? "file" : "files"}.`);
   164→    } catch (e) {
   165→      setConnectError((e as Error).message);
   166→      toast.error((e as Error).message);
   167→    } finally {
   168→      setUploading(false);
   169→    }
   170→  }
   171→
   172→  function onFileInput(e: React.ChangeEvent<HTMLInputElement>) {
   173→    const files = e.target.files;
   174→    if (files && files.length > 1) {
   175→      void handleUploadMultiple(files);
   176→    } else if (files && files.length === 1) {
   177→      void handleUpload(files[0]);
   178→    }
   179→  }
   180→
   181→  function onDrop(e: React.DragEvent) {
   182→    e.preventDefault();
   183→    setDragActive(false);
   184→    const files = e.dataTransfer.files;
   185→    if (files && files.length) void handleUploadMultiple(files);
   186→  }

   188→  async function reopenSession(s: RecentSession) {
   189→    setLoadingSession(s.id);
   190→    try {
   191→      const res = await fetch(`/api/sessions/${s.id}`);
   192→      if (!res.ok) throw new Error("Session not found");
   193→      const data = await res.json();
   194→      loadSession({
   195→        sessionId: data.sessionId,
   196→        mode: data.mode === "classic" ? "classic" : "sql",
   197→        label: data.label,
   198→        dialect: data.dialect,
   199→        schema: data.schemaSnapshot,
   200→        canWrite: data.canWrite,
   201→        zenMode: data.zenMode,
   202→        suggestedQuestions: data.suggestedQuestions || [],
   203→        messages: (data.messages || []).map((m: { role: string; content: string; id?: string }, i: number) => ({
   204→          id: m.id || `msg-${i}`,
   205→          role: m.role as "user" | "assistant" | "system",
   206→          content: m.content,
   207→        })),
   208→        canvas: data.canvas || [],
   209→      });
   210→      toast.success(`Reopened "${s.label}"`);
   211→    } catch (e) {
   212→      toast.error((e as Error).message);
   213→    } finally {
   214→      setLoadingSession(null);
   215→    }
   216→  }

   523→                className="w-full flex items-center gap-1.5 px-1 py-1.5 text-[11px] font-medium text-muted-foreground/70 uppercase tracking-wide hover:text-foreground transition-colors"

   526→                Recent sessions
   527→                <span className="text-muted-foreground/50 normal-case tracking-normal">({recent.length})</span>
   528→                <ChevronRight className={cn("h-3 w-3 ml-auto transition-transform", showRecent && "rotate-90")} />
   529→              </button>
   530→              <AnimatePresence>
   531→                {showRecent && (
   532→                  <motion.div
   533→                    initial={{ opacity: 0, height: 0 }}
   534→                    animate={{ opacity: 1, height: "auto" }}
   535→                    exit={{ opacity: 0, height: 0 }}
   536→                    className="overflow-hidden"
   537→                  >
   538→                    <div className="space-y-1.5 pt-1">
   539→                      {recent.map((s) => (
   540→                        <motion.button
   541→                          key={s.id}
   542→                          onClick={() => reopenSession(s)}
   543→                          disabled={loadingSession === s.id}
   544→                          whileHover={{ x: 2 }}
   545→                          className="group w-full flex items-center gap-2.5 rounded-xl border border-border bg-card/60 backdrop-blur px-3 py-2.5 text-left hover:border-primary/30 hover:bg-accent/30 transition-all disabled:opacity-60"
   546→                        >
   547→                          <div className={cn(
   548→                            "h-7 w-7 rounded-lg border flex items-center justify-center shrink-0",
   549→                            s.mode === "classic" ? "bg-primary/8 border-primary/15" : "bg-primary/8 border-primary/15"
   550→                          )}>
   551→                            {s.mode === "classic"
   552→                              ? <FileSpreadsheet className="h-3.5 w-3.5 text-primary" />
   553→                              : <Database className="h-3.5 w-3.5 text-primary" />}
   554→                          </div>
   555→                          <div className="min-w-0 flex-1">
   556→                            <div className="text-sm font-medium truncate">{s.label}</div>
   557→                            <div className="flex items-center gap-1.5 mt-0.5">
   558→                              <span className="text-[9px] uppercase tracking-wide text-muted-foreground bg-muted/60 rounded px-1 py-0.5">{s.dialect}</span>
   559→                              {s.mode === "classic" && (
   560→                                <span className="text-[9px] uppercase tracking-wide text-primary bg-primary/10 rounded px-1 py-0.5">Classic</span>
   561→                              )}
   562→                              {s.zenMode && (
   563→                                <span className="text-[9px] uppercase tracking-wide text-amber-600 dark:text-amber-400 bg-amber-500/10 rounded px-1 py-0.5">Zen</span>
   564→                              )}
   565→                              <span className="text-[10px] text-muted-foreground/60 tabular-nums">{timeAgo(s.updatedAt)}</span>
   566→                            </div>
   567→                          </div>
   568→                          <span
   569→                            onClick={(e) => deleteSession(s.id, e)}
   570→                            role="button"
   571→                            tabIndex={0}
   572→                            className="opacity-0 group-hover:opacity-100 text-muted-foreground hover:text-destructive transition-all p-1 rounded cursor-pointer"
   573→                            title="Delete session"
   574→                          >
   575→                            <Trash2 className="h-3.5 w-3.5" />
   576→                          </span>
   577→                          {loadingSession === s.id ? (
   578→                            <Loader2 className="h-4 w-4 animate-spin text-primary shrink-0" />
   579→                          ) : (
   580→                            <ChevronRight className="h-4 w-4 text-muted-foreground group-hover:text-primary transition-colors shrink-0" />
   581→                          )}
   582→                        </motion.button>
   583→                      ))}
   584→                    </div>
   585→                  </motion.div>
   586→                )}
   587→              </AnimatePresence>
   588→            </motion.div>
   589→          )}
   590→        </div>
   591→      </main>
   592→
   593→      <footer className="relative border-t border-border/40 py-3.5 px-6 flex items-center justify-between text-[11px] text-muted-foreground/60">
   594→        <span>Built with Next.js · Vega-Lite · an OpenAI-compatible LLM</span>
   595→        <span className="hidden sm:flex items-center gap-1.5">
   596→          <Sparkles className="h-3 w-3" />
   597→          Agentic data analysis
   598→        </span>
   599→      </footer>
   600→    </div>
   601→  );
   602→}

   604→function timeAgo(iso: string): string {
   605→  const diff = Date.now() - new Date(iso).getTime();
   606→  const mins = Math.floor(diff / 60000);
   607→  if (mins < 1) return "now";
   608→  if (mins < 60) return `${mins}m ago`;
   609→  const hrs = Math.floor(mins / 60);
   610→  if (hrs < 24) return `${hrs}h ago`;
   611→  const days = Math.floor(hrs / 24);
   612→  return `${days}d ago`;
   613→}
