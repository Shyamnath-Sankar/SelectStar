     1→/**
     2→ * Connection layer — dialect-agnostic abstraction over SQLite and PostgreSQL.
     3→ *
     4→ * This is the unglamorous foundation everything else depends on (spec §4).
     5→ * Each dialect implements the `DbConnection` interface. Adding MySQL later
     6→ * means implementing this interface once — agent logic never touches drivers.
     7→ *
     8→ * NOTE: This module is server-only. It imports native drivers that must not
     9→ * be bundled into client code. Import it only from API routes / agents.
    10→ */
    11→import type { Dialect, SchemaSnapshot, TableInfo, ColumnInfo } from "@/lib/types";
    12→import type Database from "better-sqlite3";
    13→import { getClassicConnection, reopenClassicFromDisk } from "@/lib/classic-registry";
    14→
    15→// ---------------------------------------------------------------------------
    16→// Public interface
    17→// ---------------------------------------------------------------------------
    18→
    19→export interface QueryResult {
    20→  columns: { name: string; dtype?: string }[];
    21→  rows: Record<string, unknown>[];
    22→  rowCount: number;
    23→  durationMs: number;
    24→}
    25→
    26→export interface WriteResult {
    27→  rowsAffected: number;
    28→  durationMs: number;
    29→}
    30→
    31→export interface DbConnection {
    32→  readonly dialect: Dialect;
    33→  /**
    34→   * Verify the connection is actually live (DNS resolved, authenticated,
    35→   * server reachable). For lazy drivers like pg's Pool this forces a real
    36→   * round-trip. Throws on connection-level failure (ENOTFOUND, ECONNREFUSED,
    37→   * authentication, SSL, etc.) — distinct from SQL/introspection errors.
    38→   */
    39→  ping(): Promise<void>;
    40→  /** Full introspection pass (spec §4.3). */
    41→  introspect(): Promise<SchemaSnapshot>;
    42→  /** Run a read query. Throws on error. */
    43→  query(sql: string, limit?: number): Promise<QueryResult>;
    44→  /**
    45→   * Run a write statement inside a transaction. Returns rowsAffected.
    46→   * Callers are responsible for confirmation gating (Zen mode) and logging.
    47→   * If `rollback` is true, the statement runs but is immediately rolled back
    48→   * so the user can see the affected count without committing (spec §6.4).
    49→   */
    50→  executeWrite(sql: string, opts?: { rollback?: boolean }): Promise<WriteResult>;
    51→  /** Best-effort detection of whether the role can write. */
    52→  detectCanWrite(): Promise<boolean>;
    53→  /** Generate a few suggested starter questions from the schema. */
    54→  close(): void;
    55→}

    57→// ---------------------------------------------------------------------------
    58→// Connection string parsing + factory
    59→// ---------------------------------------------------------------------------
    60→
    61→export interface ParsedConnection {
    62→  dialect: Dialect;
    63→  /** Normalised connection string safe to log (passwords masked). */
    64→  safeString: string;
    65→  /** The real connection string passed to the driver. */
    66→  raw: string;
    67→}
    68→
    69→const DEMO_LABEL = "demo";
    70→
    71→/**
    72→ * Parse a connection string. Supports:
    73→ *  - "demo" → the bundled demo SQLite database (db/demo.db)
    74→ *  - "sqlite:<path>" or "file:<path>" → a SQLite file
    75→ *  - "postgresql://..." / "postgres://..." → PostgreSQL via pg
    76→ *  - "mysql://..." → MySQL (not yet implemented, surfaces a clear error)
    77→ */
    78→export function parseConnectionString(input: string): ParsedConnection {
    79→  const trimmed = input.trim();
    80→  if (!trimmed) throw new Error("Connection string is empty.");
    81→
    82→  // Demo shorthand.
    83→  if (trimmed.toLowerCase() === DEMO_LABEL) {
    84→    return { dialect: "sqlite", safeString: "demo://demo.db", raw: "db/demo.db" };
    85→  }
    86→
    87→  const lower = trimmed.toLowerCase();
    88→  if (lower.startsWith("postgres")) {
    89→    return { dialect: "postgres", safeString: maskPassword(trimmed), raw: trimmed };
    90→  }
    91→  if (lower.startsWith("mysql://")) {
    92→    throw new Error(
    93→      "MySQL support is wired into the architecture but the driver isn't installed in this build. Try PostgreSQL or SQLite, or use the demo database."
    94→    );
    95→  }
    96→  // Bare file path or sqlite:/file / file:file
    97→  if (lower.startsWith("sqlite:") || lower.startsWith("file:") || lower.startsWith("/") || lower.startsWith("./") || /^[a-z]:\\/i.test(trimmed)) {
    98→    const path = trimmed.replace(/^(sqlite:|file:)/i, "");
    99→    return { dialect: "sqlite", safeString: `sqlite:${path}`, raw: path };
   100→  }
   101→  throw new Error(
   102→    `Unrecognised connection string. Use "demo", a SQLite file path, or a postgresql:// URL. Got: "${trimmed.slice(0, 60)}"`
   103→  );
   104→}
   105→
   106→function maskPassword(connStr: string): string {
   107→  try {
   108→    return connStr.replace(/(:\/\/[^:]+:)[^@]+@/, "$1••••@");
   109→  } catch {
   110→    return connStr;
   111→  }
   112→}

   114→// ---------------------------------------------------------------------------
   115→// In-memory connection registry — keyed by sessionId.
   116→// A real deployment would use a connection pool / external cache; in v1 the
   117→// spec explicitly says in-memory is fine.
   118→// ---------------------------------------------------------------------------
   119→
   120→const connections = new Map<string, DbConnection>();
   121→
   122→export async function openConnection(sessionId: string, connStr: string): Promise<DbConnection> {
   123→  // Reuse an existing open connection for this session.
   124→  const existing = connections.get(sessionId);
   125→  if (existing) return existing;
   126→
   127→  // Classic mode: the connection string is a "classic://<path>" URI pointing
   128→  // at the on-disk copy of the uploaded file. Re-hydrate the in-memory dataset.
   129→  if (connStr.startsWith("classic://")) {
   130→    const filePath = connStr.slice("classic://".length);
   131→    const conn = await reopenClassicFromDisk(sessionId, filePath);
   132→    if (!conn) {
   133→      throw new Error(
   134→        "Couldn't re-open this Classic dataset — the uploaded file may have been removed. Please re-upload the CSV/XLSX."
   135→      );
   136→    }
   137→    connections.set(sessionId, conn);
   138→    return conn;
   139→  }
   140→
   141→  const parsed = parseConnectionString(connStr);
   142→  let conn: DbConnection;
   143→  if (parsed.dialect === "sqlite") {
   144→    conn = new SqliteConnection(parsed.raw);
   145→  } else if (parsed.dialect === "postgres") {
   146→    conn = new PostgresConnection(parsed.raw);
   147→  } else {
   148→    throw new Error(`Dialect ${parsed.dialect} is not implemented yet.`);
   149→  }
   150→  connections.set(sessionId, conn);
   151→  return conn;
   152→}
   153→
   154→export function getConnection(sessionId: string): DbConnection | undefined {
   155→  // Check the live-DB registry first; fall back to the classic in-memory
   156→  // registry (classic connections are NOT stored in `connections` because
   157→  // they're created via /api/classic/upload which calls registerClassicDataset
   158→  // directly).
   159→  return connections.get(sessionId) ?? getClassicConnection(sessionId);
   160→}
   161→
   162→export function closeConnection(sessionId: string): void {
   163→  const conn = connections.get(sessionId);
   164→  if (conn) {
   165→    try { conn.close(); } catch { /* ignore */ }
   166→    connections.delete(sessionId);
   167→  }
   168→}

   170→// ---------------------------------------------------------------------------
   171→// SQLite connection (better-sqlite3, synchronous — wrapped in promises)
   172→// ---------------------------------------------------------------------------
   173→
   174→class SqliteConnection implements DbConnection {
   175→  readonly dialect: Dialect = "sqlite";
   176→  private db: Database.Database;
   177→
   178→  constructor(path: string) {
   179→    // Imported lazily so client bundles never pull the native module.
   180→    // eslint-disable-next-line @typescript-eslint/no-require-imports
   181→    const Database = require("better-sqlite3");
   182→    if (path === "db/demo.db" || path === "demo") {
   183→      this.db = new Database("db/demo.db", { readonly: false, fileMustExist: true });
   184→    } else {
   185→      this.db = new Database(path, { readonly: false });
   186→    }
   187→  }
   188→
   189→  // SQLite opens the file synchronously in the constructor, so by the time
   190→  // we get here the connection is already established (or it threw).
   191→  async ping(): Promise<void> {
   192→    this.db.prepare("SELECT 1").get();
   193→  }
   194→
   195→  async introspect(): Promise<SchemaSnapshot> {
   196→    const tables = this.db.prepare(
   197→      `SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name`
   198→    ).all() as { name: string }[];
   199→
   200→    const tableInfos: TableInfo[] = [];
   201→    for (const { name } of tables) {
   202→      const cols = this.db.prepare(`PRAGMA table_info(${quoteIdent(name)})`).all() as {
   203→        cid: number; name: string; type: string; notnull: number; pk: number;
   204→      }[];
   205→      const fks = this.db.prepare(`PRAGMA foreign_key_list(${quoteIdent(name)})`).all() as {
   206→        from: string; table: string; to: string;
   207→      }[];
   208→      const fkMap = new Map<string, { table: string; column: string }>();
   209→      for (const fk of fks) fkMap.set(fk.from, { table: fk.table, column: fk.to });
   210→
   211→      let rowCount = 0;
   212→      try {
   213→        rowCount = (this.db.prepare(`SELECT COUNT(*) c FROM ${quoteIdent(name)}`).get() as { c: number }).c;
   214→      } catch { /* view without rows */ }
   215→
   216→      tableInfos.push({
   217→        name,
   218→        columns: cols.map((c) => {
   219→          const fk = fkMap.get(c.name);
   220→          return {
   221→            name: c.name,
   222→            dataType: c.type || "TEXT",
   223→            nullable: c.notnull === 0,
   224→            isPrimaryKey: c.pk > 0,
   225→            isForeignKey: !!fk,
   226→            references: fk ?? null,
   227→          } satisfies ColumnInfo;
   228→        }),
   229→        rowCount,
   230→        description: inferTableDescription(name),
   231→      });
   232→    }
   233→
   234→    return {
   235→      dialect: "sqlite",
   236→      database: "demo.db",
   237→      tables: tableInfos,
   238→      introspectedAt: new Date().toISOString(),
   239→    };
   240→  }
   241→
   242→  async query(sql: string, limit = 500): Promise<QueryResult> {
   243→    const start = Date.now();
   244→    const stmt = this.db.prepare(sql);
   245→    // Apply a guard LIMIT for SELECT if the user didn't supply one.
   246→    const guarded = applyRowLimit(sql, limit);
   247→    const rows = (this.db.prepare(guarded).all() as Record<string, unknown>[]);
   248→    const durationMs = Date.now() - start;
   249→    // Derive columns from the first row, falling back to stmt.columns() if available.
   250→    let columns: { name: string; dtype?: string }[] = [];
   251→    try {
   252→      const stmtCols = this.db.prepare(guarded).columns?.() as { name: string; type?: string }[] | undefined;
   253→      if (stmtCols && stmtCols.length) columns = stmtCols.map((c) => ({ name: c.name, dtype: c.type }));
   254→    } catch { /* columns() not available */ }
   255→    if (!columns.length && rows.length) {
   256→      columns = Object.keys(rows[0]).map((name) => ({ name }));
   257→    }
   258→    return { columns, rows, rowCount: rows.length, durationMs };
   259→  }
   260→
   261→  async executeWrite(sql: string, opts?: { rollback?: boolean }): Promise<WriteResult> {
   262→    const start = Date.now();
   263→    let rowsAffected = 0;
   264→    // Manual BEGIN/COMMIT|ROLLBACK so we can dry-run a write and discard it.
   265→    this.db.exec("BEGIN");
   266→    try {
   267→      const r = this.db.prepare(sql).run();
   268→      rowsAffected = r.changes;
   269→      if (opts?.rollback) this.db.exec("ROLLBACK");
   270→      else this.db.exec("COMMIT");
   271→    } catch (e) {
   272→      try { this.db.exec("ROLLBACK"); } catch { /* ignore */ }
   273→      throw e;
   274→    }
   275→    return { rowsAffected: rowsAffected ?? 0, durationMs: Date.now() - start };
   276→  }
   277→
   278→  async detectCanWrite(): Promise<boolean> {
   279→    try {
   280→      // Attempt a no-op write capability check: try to create a temp table.
   281→      this.db.exec("CREATE TABLE IF NOT EXISTS __zen_probe (id INTEGER)");
   282→      this.db.exec("DROP TABLE __zen_probe");
   283→      return true;
   284→    } catch {
   285→      return false;
   286→    }
   287→  }
   288→
   289→  close(): void {
   290→    try { this.db.close(); } catch { /* ignore */ }
   291→  }
   292→}

   294→// ---------------------------------------------------------------------------

   297→
   298→class PostgresConnection implements DbConnection {
   299→  readonly dialect: Dialect = "postgres";
   300→  private pool: import("pg").Pool | null = null;
   301→  private connStr: string;
   302→
   303→  constructor(connStr: string) {
   304→    this.connStr = connStr;
   305→  }
   306→
   307→  /**
   308→   * Create the Pool. Called once after ping() determines SSL support.
   309→   */
   310→  private ensurePool(useSSL: boolean) {
   311→    if (this.pool) return;
   312→    // eslint-disable-next-line @typescript-eslint/no-require-imports
   313→    const { Pool } = require("pg");
   314→    this.pool = new Pool({
   315→      connectionString: this.connStr,
   316→      max: 4,
   317→      idleTimeoutMillis: 30000,
   318→      connectionTimeoutMillis: 12000,
   319→      ssl: useSSL ? { rejectUnauthorized: false } : false,
   320→    });
   321→  }
   322→
   323→  private async getClient() {
   324→    if (!this.pool) throw new Error("Connection not established — call ping() first.");
   325→    return this.pool.connect();
   326→  }
   327→
   328→  /**
   329→   * Probe the server with a disposable pg.Client to verify connectivity AND
   330→   * detect SSL support, then build the Pool once with the correct setting.
   331→   *
   332→   * Why Client instead of Pool?  Pool.connect() is lazy and manages its own
   333→   * internal queue — rapidly destroying and recreating Pools (SSL → no-SSL
   334→   * fallback) leaves zombie connections and hangs on some OS/driver combos.
   335→   * A bare Client is a single socket with no lifecycle baggage.
   336→   */
   337→  async ping(): Promise<void> {
   338→    const TIMEOUT_MS = 15_000;
   339→
   340→    const probe = async (useSSL: boolean): Promise<void> => {
   341→      // eslint-disable-next-line @typescript-eslint/no-require-imports
   342→      const { Client } = require("pg");
   343→      const client = new Client({
   344→        connectionString: this.connStr,
   345→        connectionTimeoutMillis: 10_000,
   346→        ssl: useSSL ? { rejectUnauthorized: false } : false,
   347→      });
   348→      try {
   349→        await client.connect();
   350→        await client.query("SELECT 1");
   351→      } finally {
   352→        client.end().catch(() => {});
   353→      }
   354→    };
   355→
   356→    const withTimeout = <T>(p: Promise<T>): Promise<T> =>
   357→      Promise.race([
   358→        p,
   359→        new Promise<never>((_, reject) =>
   360→          setTimeout(
   361→            () => reject(new Error(
   362→              `Connection timed out after ${TIMEOUT_MS / 1000}s — the host may be unreachable or a firewall is blocking port 5432`
   363→            )),
   364→            TIMEOUT_MS
   365→          )
   366→        ),
   367→      ]);
   368→
   369→    // Determine whether the user explicitly set ssl in the connection string.
   370→    const userSetSSL = /[?&]ssl(mode)?=/i.test(this.connStr);
   371→
   372→    if (userSetSSL) {
   373→      // Respect the user's explicit setting — don't second-guess it.
   374→      await withTimeout(probe(false)); // pg parses sslmode from the URL itself
   375→      this.ensurePool(false);          // let the URL's sslmode drive the Pool too
   376→      return;
   377→    }
   378→
   379→    // No explicit ssl param → try SSL first, fall back to plain.
   380→    try {
   381→      console.log("[pg] Probing with SSL (rejectUnauthorized:false)…");
   382→      await withTimeout(probe(true));
   383→      console.log("[pg] SSL probe succeeded.");
   384→      this.ensurePool(true);
   385→    } catch (e) {
   386→      const msg = (e as Error).message ?? "";
   387→      const isSSLError = /ssl|tls|certificate|EPROTO|self.signed|does not support/i.test(msg);
   388→      if (isSSLError) {
   389→        console.log("[pg] SSL not supported, retrying without SSL…");
   390→        try {
   391→          await withTimeout(probe(false));
   392→          console.log("[pg] Plain connection succeeded.");
   393→          this.ensurePool(false);
   394→        } catch (e2) {
   395→          throw e2; // non-SSL also failed — throw that error
   396→        }
   397→      } else {
   398→        throw e; // not an SSL issue — auth, DNS, timeout, etc.
   399→      }
   400→    }
   401→  }
   402→
   403→  async introspect(): Promise<SchemaSnapshot> {
   404→    const MAX_TABLES = 200;
   405→    const TIMEOUT_MS = 30_000;
   406→    const client = await this.getClient();
   407→
   408→    const timeout = setTimeout(() => {
   409→      console.warn("[pg] Introspection approaching timeout — releasing client.");
   410→      client.release(true); // force-destroy
   411→    }, TIMEOUT_MS);
   412→
   413→    try {
   414→      console.log("[pg] Introspecting schema…");
   415→
   416→      // 1. Get tables (capped).
   417→      const tablesRes = await client.query<{
   418→        table_schema: string; table_name: string;
   419→      }>(`
   420→        SELECT table_schema, table_name
   421→        FROM information_schema.tables
   422→        WHERE table_schema NOT IN ('pg_catalog','information_schema')
   423→          AND table_type = 'BASE TABLE'
   424→        ORDER BY table_schema, table_name
   425→        LIMIT ${MAX_TABLES}
   426→      `);
   427→      console.log(`[pg] Found ${tablesRes.rows.length} tables (cap: ${MAX_TABLES}).`);
   428→
   429→      if (tablesRes.rows.length === 0) {
   430→        return { dialect: "postgres", database: undefined, tables: [], introspectedAt: new Date().toISOString() };
   431→      }
   432→
   433→      // Build a schema/table pair filter for batch queries.
   434→      const pairs = tablesRes.rows;
   435→      const schemaNames = [...new Set(pairs.map(p => p.table_schema))];
   436→      const tableNames = [...new Set(pairs.map(p => p.table_name))];
   437→
   438→      // 2. Batch columns — one query for ALL tables.
   439→      const colsMap = new Map<string, { column_name: string; data_type: string; is_nullable: string }[]>();
   440→      try {
   441→        const colsRes = await client.query<{
   442→          table_schema: string; table_name: string;
   443→          column_name: string; data_type: string; is_nullable: string;
   444→        }>(`
   445→          SELECT table_schema, table_name, column_name, data_type, is_nullable
   446→          FROM information_schema.columns
   447→          WHERE table_schema = ANY($1) AND table_name = ANY($2)
   448→          ORDER BY table_schema, table_name, ordinal_position
   449→        `, [schemaNames, tableNames]);
   450→        for (const r of colsRes.rows) {
   451→          const key = `${r.table_schema}.${r.table_name}`;
   452→          if (!colsMap.has(key)) colsMap.set(key, []);
   453→          colsMap.get(key)!.push(r);
   454→        }
   455→      } catch (e) {
   456→        console.warn("[pg] Batch column query failed:", (e as Error).message);
   457→      }
   458→      console.log("[pg] Columns fetched.");
   459→
   460→      // 3. Batch primary keys.
   461→      const pkMap = new Map<string, Set<string>>();
   462→      try {
   463→        const pkRes = await client.query<{
   464→          table_schema: string; table_name: string; column_name: string;
   465→        }>(`
   466→          SELECT tc.table_schema, tc.table_name, kcu.column_name
   467→          FROM information_schema.table_constraints tc
   468→          JOIN information_schema.key_column_usage kcu
   469→            ON tc.constraint_name = kcu.constraint_name AND tc.table_schema = kcu.table_schema
   470→          WHERE tc.constraint_type = 'PRIMARY KEY'
   471→            AND tc.table_schema = ANY($1) AND tc.table_name = ANY($2)
   472→        `, [schemaNames, tableNames]);
   473→        for (const r of pkRes.rows) {
   474→          const key = `${r.table_schema}.${r.table_name}`;
   475→          if (!pkMap.has(key)) pkMap.set(key, new Set());
   476→          pkMap.get(key)!.add(r.column_name);
   477→        }
   478→      } catch { /* permission denied */ }
   479→      console.log("[pg] Primary keys fetched.");
   480→
   481→      // 4. Batch foreign keys.
   482→      const fkMap = new Map<string, Map<string, { table: string; column: string }>>();
   483→      try {
   484→        const fkRes = await client.query<{
   485→          table_schema: string; table_name: string;
   486→          column_name: string; ref_table: string; ref_column: string;
   487→        }>(`
   488→          SELECT tc.table_schema, tc.table_name,
   489→                 kcu.column_name, ccu.table_name AS ref_table, ccu.column_name AS ref_column
   490→          FROM information_schema.table_constraints tc
   491→          JOIN information_schema.key_column_usage kcu
   492→            ON tc.constraint_name = kcu.constraint_name AND tc.table_schema = kcu.table_schema
   493→          JOIN information_schema.constraint_column_usage ccu
   494→            ON tc.constraint_name = ccu.constraint_name AND tc.table_schema = ccu.constraint_schema
   495→          WHERE tc.constraint_type = 'FOREIGN KEY'
   496→            AND tc.table_schema = ANY($1) AND tc.table_name = ANY($2)
   497→        `, [schemaNames, tableNames]);
   498→        for (const r of fkRes.rows) {
   499→          const key = `${r.table_schema}.${r.table_name}`;
   500→          if (!fkMap.has(key)) fkMap.set(key, new Map());
   501→          fkMap.get(key)!.set(r.column_name, { table: r.ref_table, column: r.ref_column });
   502→        }
   503→      } catch { /* permission denied */ }
   504→      console.log("[pg] Foreign keys fetched.");
   505→
   506→      // 5. Approximate row counts from pg_stat — instant, no COUNT(*).
   507→      const rowCountMap = new Map<string, number>();
   508→      try {
   509→        const rcRes = await client.query<{
   510→          schemaname: string; relname: string; n_live_tup: string;
   511→        }>(`
   512→          SELECT schemaname, relname, n_live_tup
   513→          FROM pg_stat_user_tables
   514→          WHERE schemaname = ANY($1) AND relname = ANY($2)
   515→        `, [schemaNames, tableNames]);
   516→        for (const r of rcRes.rows) {
   517→          rowCountMap.set(`${r.schemaname}.${r.relname}`, parseInt(r.n_live_tup, 10) || 0);
   518→        }
   519→      } catch { /* permission denied — leave 0 */ }
   520→      console.log("[pg] Row counts fetched.");
   521→
   522→      // 6. Assemble results.
   523→      const tableInfos: TableInfo[] = [];
   524→      for (const t of pairs) {
   525→        const key = `${t.table_schema}.${t.table_name}`;
   526→        const cols = colsMap.get(key);
   527→        if (!cols || cols.length === 0) continue; // no column info — skip
   528→
   529→        const pks = pkMap.get(key) ?? new Set<string>();
   530→        const fks = fkMap.get(key) ?? new Map<string, { table: string; column: string }>();
   531→
   532→        tableInfos.push({
   533→          name: t.table_name,
   534→          schema: t.table_schema,
   535→          columns: cols.map((c) => {
   536→            const fk = fks.get(c.column_name);
   537→            return {
   538→              name: c.column_name,
   539→              dataType: c.data_type,
   540→              nullable: c.is_nullable === "YES",
   541→              isPrimaryKey: pks.has(c.column_name),
   542→              isForeignKey: !!fk,
   543→              references: fk ?? null,
   544→            };
   545→          }),
   546→          rowCount: rowCountMap.get(key) ?? 0,
   547→          description: inferTableDescription(t.table_name),
   548→        });
   549→      }
   550→
   551→      console.log(`[pg] Introspection complete: ${tableInfos.length} tables.`);
   552→      return {
   553→        dialect: "postgres",
   554→        database: undefined,
   555→        tables: tableInfos,
   556→        introspectedAt: new Date().toISOString(),
   557→      };
   558→    } finally {
   559→      clearTimeout(timeout);
   560→      client.release();
   561→    }
   562→  }
   563→
   564→  async query(sql: string, limit = 500): Promise<QueryResult> {
   565→    const client = await this.getClient();
   566→    try {
   567→      const start = Date.now();
   568→      const guarded = applyRowLimit(sql, limit);
   569→      const res = await client.query(guarded);
   570→      return {
   571→        columns: res.fields.map((f) => ({ name: f.name, dtype: dataTypeName(f.dataTypeID) })),
   572→        rows: res.rows as Record<string, unknown>[],
   573→        rowCount: res.rowCount ?? res.rows.length,
   574→        durationMs: Date.now() - start,
   575→      };
   576→    } finally {
   577→      client.release();
   578→    }
   579→  }
   580→
   581→  async executeWrite(sql: string, opts?: { rollback?: boolean }): Promise<WriteResult> {
   582→    const client = await this.getClient();
   583→    try {
   584→      const start = Date.now();
   585→      await client.query("BEGIN");
   586→      let rowsAffected = 0;
   587→      try {
   588→        const res = await client.query(sql);
   589→        rowsAffected = res.rowCount ?? 0;
   590→        if (opts?.rollback) await client.query("ROLLBACK");
   591→        else await client.query("COMMIT");
   592→      } catch (e) {
   593→        await client.query("ROLLBACK");
   594→        throw e;
   595→      }
   596→      return { rowsAffected, durationMs: Date.now() - start };
   597→    } finally {
   598→      client.release();
   599→    }
   600→  }
   601→
   602→  async detectCanWrite(): Promise<boolean> {
   603→    const client = await this.getClient();
   604→    try {
   605→      // Use has_database_privilege for a database-level check (safer than
   606→      // has_table_privilege which needs a specific table name).
   607→      const res = await client.query(
   608→        `SELECT has_database_privilege(current_user, current_database(), 'CREATE') AS can_write`
   609→      );
   610→      return res.rows[0]?.can_write === true;
   611→    } catch {
   612→      // Fallback: assume read-only; the user can enable Zen mode manually.
   613→      return false;
   614→    } finally {
   615→      client.release();
   616→    }
   617→  }
   618→
   619→  close(): void {
   620→    try { if (this.pool) void this.pool.end(); } catch { /* ignore */ }
   621→  }
   622→}

   628→function quoteIdent(name: string): string {
   629→  return `"${name.replace(/"/g, '""')}"`;
   630→}
   631→
   632→/**
   633→ * Wrap a SELECT so it never returns more than `limit` rows. If the query
   634→ * already has a LIMIT clause we leave it alone (but cap it at limit*2 to
   635→ * avoid runaway queries). Non-SELECT statements are returned unchanged.
   636→ */
   637→export function applyRowLimit(sql: string, limit: number): string {
   638→  const trimmed = sql.trim().replace(/;$/, "");
   639→  const isSelect = /^\s*\(?\s*select/i.test(trimmed);
   640→  if (!isSelect) return trimmed;
   641→  const hasLimit = /\blimit\b/i.test(trimmed);
   642→  if (hasLimit) return trimmed;
   643→  return `${trimmed} LIMIT ${limit}`;
   644→}
   645→
   646→/** Heuristic: is this SQL a write statement? (INSERT/UPDATE/DELETE/DDL etc.) */
   647→export function isWriteStatement(sql: string): boolean {
   648→  return /^\s*(insert|update|delete|drop|alter|create|truncate|merge|grant|revoke)\b/i.test(sql.trim());
   649→}

   651→function inferTableDescription(name: string): string {
   652→  const n = name.toLowerCase();
   653→  if (n.includes("order")) return "Customer orders.";
   654→  if (n.includes("customer")) return "Customer records.";
   655→  if (n.includes("product")) return "Product catalog.";
   656→  if (n.includes("categor")) return "Product categories.";
   657→  if (n.includes("region")) return "Geographic regions.";
   658→  if (n.includes("item")) return "Line items belonging to orders.";
   659→  return undefined;
   660→}
   661→
   662→function dataTypeName(oid: number | undefined): string | undefined {
   663→  if (oid === undefined) return undefined;
   664→  // A tiny subset; pg returns numeric OIDs.
   665→  const map: Record<number, string> = {
   666→    16: "boolean", 17: "bytea", 20: "int8", 21: "int2", 23: "int4",
   667→    25: "text", 700: "float4", 701: "float8", 1043: "varchar",
   668→    1082: "date", 1114: "timestamp", 1184: "timestamptz", 1700: "numeric",
   669→  };
   670→  return map[oid];
   671→}
